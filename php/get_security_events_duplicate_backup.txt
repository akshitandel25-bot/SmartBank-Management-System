
<?php
header("Content-Type: application/json; charset=UTF-8");

require_once "db.php";

$accountId = filter_input(
    INPUT_GET, "account_id", FILTER_VALIDATE_INT
);

if (!$accountId || $accountId < 1) {
    http_response_code(400);
    echo json_encode([
        "success" => false,
        "message" => "A valid account ID is required."
    ]);
    exit;
}

$sql = "SELECT event_id, account_id, transaction_id,
               event_type, risk_level, description,
               recommended_action, event_status, created_at
        FROM security_events
        WHERE account_id = ?
        ORDER BY created_at DESC, event_id DESC";

$stmt = $conn->prepare($sql);

if (!$stmt) {
    http_response_code(500);
    echo json_encode([
        "success" => false,
        "message" => "Could not prepare security event query."
    ]);
    exit;
}

$stmt->bind_param("i", $accountId);
$stmt->execute();

$result = $stmt->get_result();
$events = [];

while ($row = $result->fetch_assoc()) {
    $events[] = $row;
}

echo json_encode([
    "success" => true,
    "events" => $events
]);

$stmt->close();
$conn->close();
?>
