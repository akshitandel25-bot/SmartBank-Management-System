        expensesValue = new JLabel();

        stats.add(statCard("TOTAL BALANCE", balanceValue, "Available funds"));

    private JPanel createTransactionsPage() {
        JPanel page = pagePanel();
        page.add(heading("Transactions"));
        page.add(label("Search your demo transaction history.",
                muted, 14));
        page.add(Box.createVerticalStrut(20));

        JPanel card = whitePanel();
        card.setLayout(new BorderLayout(10, 12));

        JTextField search = new JTextField();
        search.setFont(new Font("SansSerif", Font.PLAIN, 14));
        search.setBorder(BorderFactory.createCompoundBorder(
                BorderFactory.createLineBorder(new Color(220, 225, 235)),
                new EmptyBorder(10, 12, 10, 12)));
        search.setToolTipText("Search by recipient, type or status");
        card.add(search, BorderLayout.NORTH);

        JTable table = createTransactionTable();
        TableRowSorter<DefaultTableModel> sorter =
                new TableRowSorter<>(tableModel);
        table.setRowSorter(sorter);
        card.add(new JScrollPane(table), BorderLayout.CENTER);

        search.getDocument().addDocumentListener(
                new javax.swing.event.DocumentListener() {
            private void filter() {
                String query = search.getText().trim();
                if (query.isEmpty()) {
                    sorter.setRowFilter(null);
                } else {
                    sorter.setRowFilter(
                            RowFilter.regexFilter(
                                    "(?i)" + java.util.regex.Pattern.quote(query)));
                }
            }

            public void insertUpdate(
                    javax.swing.event.DocumentEvent e) { filter(); }
            public void removeUpdate(
                    javax.swing.event.DocumentEvent e) { filter(); }
            public void changedUpdate(
                    javax.swing.event.DocumentEvent e) { filter(); }
        });

        page.add(card);
        return page;
    }

    private JTable createTransactionTable() {
        if (tableModel == null) {
            tableModel = new DefaultTableModel(
                    new String[]{"Date", "Description", "Type",
                        "Amount", "Status"}, 0) {
                @Override
                public boolean isCellEditable(int row, int column) {
                    return false;
                }
            };

            addTransaction("Salary credit", "Income", 45000, "Completed");
            addTransaction("Grocery store", "Expense", -2350, "Completed");
            addTransaction("Electricity bill", "Expense", -1800, "Completed");
            addTransaction("Online shopping", "Expense", -4200, "Completed");
            addTransaction("Mobile recharge", "Expense", -499, "Completed");
        }

        JTable table = new JTable(tableModel);
        table.setRowHeight(32);
        table.setFont(new Font("SansSerif", Font.PLAIN, 12));
        table.getTableHeader().setFont(
                new Font("SansSerif", Font.BOLD, 12));
        table.setFillsViewportHeight(true);
        return table;
    }

    private void addTransaction(
            String description, String type, double amount, String status) {
        String date = LocalDateTime.now().format(
                DateTimeFormatter.ofPattern("dd MMM yyyy, HH:mm"));

        tableModel.insertRow(0, new Object[]{
            date, description, type,
            (amount < 0 ? "- " : "+ ") + money(Math.abs(amount)),
            status
        });
    }

    private String money(double amount) {
        return String.format(java.util.Locale.US, "₹%,.2f", amount);
    }

    private void refreshOverview() {
        if (balanceValue != null) {
        balanceValue.setText(money(balance));
    }
    
    if (incomeValue != null) {
        incomeValue.setText(money(income));
    }

    if (expensesValue != null) {
        expensesValue.setText(money(expenses));
    }

    if (securityValue != null) {
        securityValue.setText(
            suspiciousCount == 0
                ? "Monitoring active"
                : suspiciousCount + " flagged transaction(s)"
        );
    }
}

    private boolean validAmount(String text) {
        try {
            double amount = Double.parseDouble(text.trim());
            return Double.isFinite(amount) && amount > 0;
        } catch (NumberFormatException ex) {
            return false;
        }
    }

    private void transferMoney() {
        JTextField recipient = new JTextField();
        JTextField amountField = new JTextField();

        JPanel form = new JPanel(new GridLayout(0, 1, 4, 6));
        form.add(new JLabel("Recipient name"));
        form.add(recipient);
        form.add(new JLabel("Transfer amount (₹)"));
        form.add(amountField);

        int choice = JOptionPane.showConfirmDialog(
                this, form, "Demo money transfer",
                JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);

        if (choice != JOptionPane.OK_OPTION) return;

        String name = recipient.getText().trim();
        String amountText = amountField.getText().trim();

        if (name.isEmpty() || !validAmount(amountText)) {
            showMessage("Enter a recipient and a valid positive amount.");
            return;
        }

        double amount = Double.parseDouble(amountText);

        if (amount > balance) {
            showMessage("Insufficient demo balance.");
            return;
        }

        String recipientKey = name.toLowerCase(
        java.util.Locale.ROOT);
        
        int previousTransfers = recipientTransferCounts.getOrDefault(
        recipientKey, 0);
        
        boolean repeatedTransfer = previousTransfers >= 1;
        boolean largeTransfer = amount >= 10000;
        
        boolean flagged = largeTransfer || repeatedTransfer;
        
        recipientTransferCounts.put(recipientKey, previousTransfers + 1);
        balance -= amount;
        expenses += amount;

        String status = flagged ? "Flagged for review" : "Completed";
        addTransaction("Transfer to " + name, "Transfer", -amount, status);

        if (flagged) {
            suspiciousCount++;
            
            String reason;
            if (largeTransfer && repeatedTransfer) {
                reason = "Large and repeated transfer";
            } 
            else if (largeTransfer) {
                reason = "Large transfer";
            } 
            else {
                reason = "Repeated transfer to the same recipient";
            }
            if (statusMessage != null) {
                statusMessage.setText("⚠ " + reason + " to " + name + ": " + money(amount) + " — review recommended.");
                statusMessage.setForeground(new Color(190, 95, 20));
            }
        } 
        else if (statusMessage != null) {
            statusMessage.setText("No rule triggered by this transfer.");
            statusMessage.setForeground(muted);
        }

        refreshOverview();
        showMessage(flagged
                ? "Demo transfer recorded and flagged for review."
                : "Demo transfer recorded successfully.");
    }

    private void depositMoney() {
        String input = JOptionPane.showInputDialog(
                this, "Enter deposit amount (₹):", "Demo deposit",
                JOptionPane.PLAIN_MESSAGE);

        if (input == null) return;
        if (!validAmount(input)) {
            showMessage("Enter a valid positive amount.");
            return;
        }

        double amount = Double.parseDouble(input.trim());
        balance += amount;
        income += amount;
        addTransaction("Cash deposit", "Deposit", amount, "Completed");
        refreshOverview();
        showMessage("Demo deposit recorded successfully.");
    }

    private void withdrawMoney() {
        String input = JOptionPane.showInputDialog(
                this, "Enter withdrawal amount (₹):", "Demo withdrawal",
                JOptionPane.PLAIN_MESSAGE);

        if (input == null) return;
        if (!validAmount(input)) {
            showMessage("Enter a valid positive amount.");
            return;
        }

        double amount = Double.parseDouble(input.trim());
        if (amount > balance) {
            showMessage("Insufficient demo balance.");
            return;
        }

        balance -= amount;
        expenses += amount;
        addTransaction("Cash withdrawal", "Withdrawal", -amount, "Completed");
        refreshOverview();
        showMessage("Demo withdrawal recorded successfully.");
    }

    private void showMessage(String message) {
        JOptionPane.showMessageDialog(
                this, message, "SmartBank",
                JOptionPane.INFORMATION_MESSAGE);
    }
